cmake_minimum_required(VERSION 3.16)
project(eng26_arm_driver)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "eng26_arm_driver serial backend requires Linux")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ament_cmake REQUIRED)
set(ROS_DEPENDENCIES rclcpp rclcpp_action rclcpp_components rcl_interfaces
  sensor_msgs std_msgs diagnostic_msgs eng26_arm_interfaces trajectory_msgs builtin_interfaces)
foreach(dep IN LISTS ROS_DEPENDENCIES)
  find_package(${dep} REQUIRED)
endforeach()
add_library(eng26_host_core SHARED src/protocol.cpp src/serial_transport.cpp src/session.cpp)
target_include_directories(eng26_host_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_options(eng26_host_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(eng26_arm_component SHARED src/arm_driver_node.cpp)
target_link_libraries(eng26_arm_component eng26_host_core)
ament_target_dependencies(eng26_arm_component ${ROS_DEPENDENCIES})
target_compile_options(eng26_arm_component PRIVATE -Wall -Wextra -Wpedantic)
rclcpp_components_register_nodes(eng26_arm_component "eng26::ArmDriverNode")
add_executable(arm_driver_node src/main.cpp)
target_link_libraries(arm_driver_node eng26_arm_component)
ament_target_dependencies(arm_driver_node rclcpp)
install(TARGETS eng26_host_core eng26_arm_component arm_driver_node
  EXPORT export_eng26_arm_driver
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_protocol test/test_protocol.cpp)
  target_link_libraries(test_protocol eng26_host_core)
  ament_add_gtest(test_session test/test_session.cpp)
  target_link_libraries(test_session eng26_host_core)
endif()
ament_export_targets(export_eng26_arm_driver HAS_LIBRARY_TARGET)
ament_export_dependencies(${ROS_DEPENDENCIES})
ament_package()
